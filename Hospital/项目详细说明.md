#### AppointmentController.java - 预约接口（核心！）

```java
@RestController
@RequestMapping("/api/appointments")
@CrossOrigin(origins = "*")
public class AppointmentController {

    @Autowired
    private AppointmentService appointmentService;

    @Autowired
    private PatientRepository patientRepository;

    @Autowired
    private DoctorRepository doctorRepository;

    @Autowired
    private DepartmentRepository departmentRepository;

    @Autowired
    private AppointmentRepository appointmentRepository;

    @GetMapping
    public List<Appointment> getMyAppointmentsByOpenid(@RequestHeader("X-Openid") String openid) {
        List<Appointment> appointments = appointmentRepository.findByOpenidOrderByCreateTimeDesc(openid);
        
        for (Appointment app : appointments) {
            if (app.getDoctorId() != null) {
                Doctor doctor = doctorRepository.findById(app.getDoctorId()).orElse(null);
                if (doctor != null) {
                    app.setDoctor(doctor);
                    if (doctor.getDeptId() != null) {
                        Department dept = departmentRepository.findById(doctor.getDeptId()).orElse(null);
                        if (dept != null) {
                            doctor.setDepartment(dept);
                        }
                    }
                }
            }
        }
        return appointments;
    }

    @PostMapping
    public Appointment createSimple(@RequestHeader("X-Openid") String openid,
                                    @RequestBody Map<String, String> request) {
        String patientId = request.get("patientId");
        String doctorId = request.get("doctorId");
        String appointmentDate = request.get("appointmentDate");
        String timeSlot = request.get("timeSlot");

        Doctor doctor = doctorRepository.findById(doctorId)
                .orElseThrow(() -> new RuntimeException("医生不存在"));

        Appointment appointment = new Appointment();
        appointment.setAppointmentId(UUID.randomUUID().toString().replace("-", ""));
        appointment.setOpenid(openid);
        appointment.setPatientId(patientId);
        appointment.setDoctorId(doctorId);
        appointment.setScheduleId("SCHEDULE_" + System.currentTimeMillis());
        appointment.setAppointmentDate(LocalDate.parse(appointmentDate));
        appointment.setTimeSlot(timeSlot);
        appointment.setStatus(1);
        appointment.setCreateTime(java.time.LocalDateTime.now());
        appointment.setDoctor(doctor);

        return appointmentRepository.save(appointment);
    }

    @PostMapping("/{appointmentId}/cancel")
    public void cancel(@PathVariable String appointmentId,
                       @RequestHeader("X-Openid") String openid) {
        appointmentService.cancelAppointmentSimple(appointmentId);
    }
}
```

```python
from flask import Flask, request, jsonify
import requests
import re
import time
from urllib.parse import quote

app = Flask(__name__)

HEADERS = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
}

def extract_emergency_phone(text):
    phone_patterns = [
        r'急诊[：:]\s*([0-9-]{7,15})',
        r'急救[：:]\s*([0-9-]{7,15})',
        r'急诊科[：:]\s*([0-9-]{7,15})',
        r'120',
    ]
    
    for pattern in phone_patterns:
        matches = re.findall(pattern, text)
        if matches:
            return matches[0] if isinstance(matches[0], str) else matches[0][0]
    return None

def search_hospital_baidu(hospital_name, address=None):
    search_query = hospital_name
    if address:
        search_query += f" {address}"
    
    search_url = f"https://www.baidu.com/s?wd={quote(search_query + ' 医院 急诊电话 预约挂号')}"
    
    try:
        response = requests.get(search_url, headers=HEADERS, timeout=10)
        response.raise_for_status()
        text = response.text
        
        result_titles = re.findall(r'<h3[^>]*>(.*?)</h3>', text, re.DOTALL)
        result_urls = re.findall(r'href=["\']([^"\']*)["\'][^>]*class=["\']c-showurl["\']', text)
        
        results = []
        for i, title in enumerate(result_titles[:5]):
            clean_title = re.sub(r'<[^>]+>', '', title)
            if clean_title:
                result = {
                    'title': clean_title,
                    'url': result_urls[i] if i < len(result_urls) else ''
                }
                results.append(result)
        
        return results
    except Exception as e:
        print(f"百度搜索失败: {e}")
        return []

def scrape_hospital_info(url, hospital_name):
    try:
        response = requests.get(url, headers=HEADERS, timeout=15)
        response.raise_for_status()
        response.encoding = 'utf-8'
        
        text = response.text
        
        emergency_phone = extract_emergency_phone(text)
        
        is_24_hours = False
        keywords_24h = ['24小时', '二十四小时', '全天', '急诊']
        for keyword in keywords_24h:
            if keyword in text:
                is_24_hours = True
                break
        
        hospital_type = '综合医院'
        if '专科' in hospital_name or '专科' in text:
            hospital_type = '专科医院'
        elif '社区' in hospital_name or '社区' in text:
            hospital_type = '社区医院'
        elif '中医' in hospital_name or '中医' in text:
            hospital_type = '中医医院'
        
        level = None
        level_keywords = ['三甲', '三级甲等', '三级乙等', '三级', '二甲', '二级甲等', '二级乙等', '二级', '一甲', '一级']
        for keyword in level_keywords:
            if keyword in hospital_name or keyword in text:
                level = keyword
                break
        
        return {
            'name': hospital_name,
            'emergencyPhone': emergency_phone,
            'is24Hours': is_24_hours,
            'type': hospital_type,
            'level': level,
            'sourceUrl': url
        }
    except Exception as e:
        print(f"爬取医院信息失败: {e}")
        return {
            'name': hospital_name,
            'emergencyPhone': None,
            'is24Hours': False,
            'type': '综合医院',
            'level': None
        }

@app.route('/api/health', methods=['GET'])
```

```javascript
App({

  checkLoginStatus() {
    const token = wx.getStorageSync('token');
    const openid = wx.getStorageSync('openid');
    if (token && openid) {
      this.globalData.token = token;
      this.globalData.openid = openid;
      this.globalData.hasPatientInfo = wx.getStorageSync('hasPatientInfo') || false;
    }
  },

  getLocation() {
    wx.getLocation({
      type: 'gcj02',
      success: (res) => {
        this.globalData.latitude = res.latitude;
        this.globalData.longitude = res.longitude;
      }
    });
  },

  login() {
    return new Promise((resolve, reject) => {
      wx.login({
        success: (loginRes) => {
          this.requestSimpleLogin(loginRes.code)
            .then(resolve)
            .catch(reject);
        },
        fail: reject
      });
    });
  },

  requestSimpleLogin(code) {
    return new Promise((resolve, reject) => {
      wx.request({
        url: this.globalData.baseUrl + '/api/wechat/login',
        method: 'POST',
        data: {
          code: code,
          nickname: '微信用户',
          avatarUrl: ''
        },
        success: (res) => {
          if (res.statusCode === 200) {
            this.globalData.token = res.data.token;
            this.globalData.openid = res.data.openid;
            wx.setStorageSync('token', res.data.token);
            wx.setStorageSync('openid', res.data.openid);
            resolve(res.data);
          } else {
            reject(res);
          }
        },
        fail: reject
      });
    });
  }
})
```

```javascript
    if (!this.data.doctor) return;
    
    wx.navigateTo({
      url: `/pages/appointment/appointment?doctorId=${this.data.doctor.doctorId}`
    });
  }
})
```

## 整体工作流程

```
用户打开小程序
    ↓
app.js 检查登录状态
    ↓
首页加载科室和医生
    ↓
用户点击医生 → doctor-detail.js 调用 /api/doctors/{id}

用户点击立即预约 → appointment.js
    ↓
用户选择科室、医生、时间、确认
    ↓
submitAppointment() 调用 /api/appointments (POST)
    ↓
Java 后端保存预约到数据库
    ↓
跳转到 my-appointments 页面
    ↓
调用 /api/appointments (GET) 获取预约列表
    ↓
显示用户的预约记录
```
