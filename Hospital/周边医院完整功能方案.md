```python
import requests

AMAP_KEY = "your_amap_api_key"

def search_nearby_hospitals(latitude, longitude, radius=50000):
    """
    搜索周边医院
    """
    url = "https://restapi.amap.com/v3/place/around"
    params = {
        "key": AMAP_KEY,
        "location": f"{longitude},{latitude}",
        "keywords": "医院",
        "types": "医疗保健服务|综合医院|专科医院",
        "radius": radius,
        "offset": 50,
        "page": 1
    }
    
    response = requests.get(url, params=params, timeout=10)
    return response.json()
```

```python
@app.route('/api/nearby/search', methods=['POST'])
def search_nearby():
    """
    增强版周边搜索
    """
    data = request.json
    latitude = data.get('latitude')
    longitude = data.get('longitude')
    radius = data.get('radius', 50000)
    
    # 1. 高德地图搜索
    hospitals = search_nearby_hospitals(latitude, longitude, radius)
    
    # 2. 并发爬取医院详情
    loop = asyncio.new_event_loop()
    asyncio.set_event_loop(loop)
    
    tasks = []
    for hospital in hospitals:
        tasks.append(crawl_hospital_details(
            hospital['name'], 
            hospital.get('url')
        ))
    
    results = loop.run_until_complete(asyncio.gather(*tasks))
    
    # 3. 合并数据
    for i, hospital in enumerate(hospitals):
        hospital.update(results[i])
    
    return jsonify(hospitals)
```

## ☕ Java 后端增强

### 1. NearbyHospitalService 扩展

**文件**：`hospital-api/src/main/java/com/hospital/service/NearbyHospitalService.java`

```java
@Service
public class NearbyHospitalService {
    
    @Autowired
    private NearbyHospitalRepository hospitalRepository;
    
    @Autowired
    private NearbyDepartmentRepository deptRepository;
    
    @Autowired
    private NearbyDoctorRepository doctorRepository;
    
    @Autowired
    private RestTemplate restTemplate;
    
    @Value("${python.service.url}")
    private String pythonServiceUrl;
    
    /**
     * 搜索并保存周边医院
     */
    public List<NearbyHospital> searchAndSaveNearbyHospitals(
            Double latitude, 
            Double longitude, 
            Integer radius) {
        
        // 1. 调用 Python 爬虫服务
        String url = pythonServiceUrl + "/api/nearby/search";
        Map<String, Object> request = new HashMap<>();
        request.put("latitude", latitude);
        request.put("longitude", longitude);
        request.put("radius", radius);
        
        List<Map<String, Object>> hospitalDataList = restTemplate.postForObject(
            url, request, List.class);
        
        // 2. 保存到数据库
        List<NearbyHospital> hospitals = new ArrayList<>();
        for (Map<String, Object> data : hospitalDataList) {
            NearbyHospital hospital = saveHospitalData(data);
            hospitals.add(hospital);
        }
        
        return hospitals;
    }
    
    /**
     * 保存医院数据（包括科室和医生）
     */
    private NearbyHospital saveHospitalData(Map<String, Object> data) {
        NearbyHospital hospital = new NearbyHospital();
        hospital.setName((String) data.get("name"));
        hospital.setAddress((String) data.get("address"));
        hospital.setPhone((String) data.get("phone"));
        hospital.setLatitude((Double) data.get("latitude"));
        hospital.setLongitude((Double) data.get("longitude"));
        hospital.setIsTempData(true); // 标记为临时数据
        
        hospital = hospitalRepository.save(hospital);
        
        // 保存科室
        List<Map<String, Object>> deptDataList = 
            (List<Map<String, Object>>) data.get("departments");
        for (Map<String, Object> deptData : deptDataList) {
            NearbyDepartment dept = new NearbyDepartment();
            dept.setHospital(hospital);
            dept.setDeptName((String) deptData.get("name"));
            deptRepository.save(dept);
        }
        
        // 保存医生
        List<Map<String, Object>> doctorDataList = 
            (List<Map<String, Object>>) data.get("doctors");
        for (Map<String, Object> doctorData : doctorDataList) {
            NearbyDoctor doctor = new NearbyDoctor();
            doctor.setHospital(hospital);
            doctor.setName((String) doctorData.get("name"));
            doctor.setTitle((String) doctorData.get("title"));
            doctorRepository.save(doctor);
        }
        
        return hospital;
    }
    
    /**
     * 删除临时数据（退出时调用）
     */
    @Scheduled(cron = "0 0 2 * * ?") // 每天凌晨2点清理
    public void cleanTempData() {
        hospitalRepository.deleteByIsTempData(true);
    }
}
```

### 2. NearbyHospitalController 扩展

**文件**：`hospital-api/src/main/java/com/hospital/controller/NearbyHospitalController.java`

```java
@RestController
@RequestMapping("/api/nearby")
public class NearbyHospitalController {
    
    @Autowired
    private NearbyHospitalService hospitalService;
    
    /**
     * 搜索周边医院（带详情）
     */
    @PostMapping("/search")
    public Result<List<NearbyHospital>> searchNearby(
            @RequestBody Map<String, Object> request) {
        
        Double latitude = (Double) request.get("latitude");
        Double longitude = (Double) request.get("longitude");
        Integer radius = (Integer) request.getOrDefault("radius", 50000);
        
        List<NearbyHospital> hospitals = 
            hospitalService.searchAndSaveNearbyHospitals(
                latitude, longitude, radius);
        
        return Result.success(hospitals);
    }
    
    /**
     * 获取医院详情（含科室和医生）
     */
    @GetMapping("/hospitals/{id}")
    public Result<NearbyHospital> getHospitalDetail(@PathVariable Long id) {
        Optional<NearbyHospital> hospital = 
            hospitalRepository.findByIdWithDetails(id);
        return hospital.map(Result::success)
                .orElse(Result.error("医院不存在"));
    }
}
```

```java
public class AppointmentService {

     * 预约周边医院
     */
    public Appointment createNearbyHospitalAppointment(
            Long patientId,
            Long nearbyHospitalId,
            Long deptId,
            Long doctorId,
            String appointmentDate,
            String timeSlot) {
        
        Appointment appointment = new Appointment();
        
        // 设置周边医院
        NearbyHospital hospital = nearbyHospitalRepository.findById(nearbyHospitalId)
                .orElseThrow(() -> new RuntimeException("医院不存在"));
        appointment.setNearbyHospital(hospital);
        appointment.setIsNearbyHospital(true);
        
        // 如果有外部预约链接，记录下来
        if (hospital.getAppointmentUrl() != null) {
            appointment.setExternalAppointmentUrl(hospital.getAppointmentUrl());
        }
        
        // 其他字段设置...
        appointment.setPatient(patientRepository.findById(patientId).orElse(null));
        appointment.setAppointmentDate(appointmentDate);
        appointment.setTimeSlot(timeSlot);
        appointment.setStatus(1);
        
        return appointmentRepository.save(appointment);
    }
}
```

## 💡 重要说明

### 关于数据爬取的法律和道德问题

⚠️ **重要提醒**：
1. 爬取医院官网数据时，请遵守 `robots.txt` 规定
2. 尊重网站的服务条款
3. 不要进行高频请求，避免给目标网站造成压力
4. 考虑使用官方 API 替代爬取（如果有的话）
5. 仅供学习和个人使用，不要用于商业用途

### 关于预约功能

⚠️ **现实情况**：
1. 大多数医院都有自己的预约系统
2. 直接通过 API 预约比较困难
3. 可以提供预约链接，引导用户到医院官网预约
4. 或者记录预约信息，作为备忘
